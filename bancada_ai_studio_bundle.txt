// 2. Diminishing Returns Curve Math
export function applyDiminishingReturns(currentVal: number, delta: number): number {
  if (delta <= 0) return Math.max(0, currentVal + delta);

  if (currentVal < 70) {
    return Math.min(100, currentVal + delta);
  } else if (currentVal < 85) {
    return Math.min(100, Math.round(currentVal + delta * 0.6));
  } else if (currentVal < 95) {
    return Math.min(100, Math.round(currentVal + delta * 0.35));
  } else {
    return Math.min(100, Math.round(currentVal + delta * 0.15));
  }
}

// 3. Create Custom Torcida with Archetype & Inheritance
export function createCustomTorcidaWithArchetype(
  torcidaName: string,
  sigla: string,
  clubName: string,
  archetypeId: ArchetypeId
): { torcida: OfficialTorcida; state: StateTrackers } {
  const normName = torcidaName.trim() || "Torcida Organizada";
  const normSigla = sigla.trim().toUpperCase() || "TOC";
  const arch = ARCHETYPES[archetypeId] || ARCHETYPES.PADRAO;

  const mapped = (teamsData as OfficialTorcida[]).find(
    (t) => t.clube.toLowerCase() === clubName.trim().toLowerCase()
  );

  let baseStats: TorcidaStats = mapped
    ? {
        contingente: mapped.contingente,
        pressao_bancada: mapped.pressao_bancada,
        poder_pista: mapped.poder_pista,
        caravana: mapped.caravana,
        autonomia_financeira: mapped.autonomia_financeira,
      }
    : {
        contingente: 60,
        pressao_bancada: 65,
        poder_pista: 60,
        caravana: 55,
        autonomia_financeira: 50,
      };

  Object.entries(arch.statModifiers).forEach(([key, mod]) => {
    const k = key as keyof TorcidaStats;
    baseStats[k] = Math.min(100, Math.max(10, baseStats[k] + (mod || 0)));
  });

  const stateTrackers: StateTrackers = {
    moral: 70 + (arch.stateModifiers.moral || 0),
    risco_mp: Math.max(0, 10 + (arch.stateModifiers.risco_mp || 0)),
    relacao_clube: 0 + (arch.stateModifiers.relacao_clube || 0),
    respeito_nacional: 50 + (arch.stateModifiers.respeito_nacional || 0),
  };

  const torcida: OfficialTorcida = {
    clube: mapped ? mapped.clube : clubName || "Clube Independente",
    torcida: normName,
    sigla: normSigla,
    tier: mapped ? mapped.tier : "B",
    contingente: baseStats.contingente,
    pressao_bancada: baseStats.pressao_bancada,
    poder_pista: baseStats.poder_pista,
    caravana: baseStats.caravana,
    autonomia_financeira: baseStats.autonomia_financeira,
    perfil_predominante: `${arch.subtitle} â€¢ ${mapped ? mapped.perfil_predominante : "Nova Torcida"}`,
    estado: mapped ? mapped.estado : "SP",
    eixo_alianca: mapped ? mapped.eixo_alianca : "NEUTRO",
  };

  return { torcida, state: stateTrackers };
}

// 7. Formula-Based Math Engine for Home and Away Games
export function calculateFormulaHomeGame(
  stats: TorcidaStats,
  state: StateTrackers,
  setup: HomeGameSetup
): KeyGameResult {
  let partyBonus = 40;
  let cost = 0;
  let mpPenalty = 0;

  if (setup.visualParty === "MOSAICO_3D") {
    partyBonus += 30;
    cost += 15000;
  } else if (setup.visualParty === "RUAS_DE_FOGO") {
    partyBonus += 20;
    cost += 8000;
    mpPenalty += 10;
  } else if (setup.visualParty === "BOBINAS_PAPEL") {
    partyBonus += 15;
    cost += 3000;
  }

  if (setup.security === "LINHA_FRENTE_PORTAO") {
    cost += 5000;
  }

  if (setup.hospitality === "CHURRASCO_ALIADOS") {
    cost += 10000;
  }

  const rngPlayer = (Math.random() - 0.5) * 30;
  const effectivePlayer = Math.round(
    stats.poder_pista * 0.35 +
      stats.caravana * 0.15 +
      state.moral * 0.2 +
      partyBonus * 0.15 +
      rngPlayer
  );

  const effectiveRival = Math.round(65 + (Math.random() - 0.5) * 30);
  const isVictory = effectivePlayer >= effectiveRival;

  const scorePower = stats.pressao_bancada * 0.5 + state.moral * 0.3 + partyBonus * 0.2 + (Math.random() - 0.5) * 20;
  const scorePlayer = scorePower > 50 ? Math.floor(Math.random() * 2 + 1) : Math.floor(Math.random() * 2);
  const scoreRival = isVictory ? Math.floor(Math.random() * scorePlayer) : Math.floor(Math.random() * 2 + scorePlayer);

  const desertion = isVictory ? Math.floor(Math.random() * 20) : Math.floor(Math.random() * 60 + 30);
  const medical = isVictory ? 0 : Math.floor(Math.random() * 2000 + 800);

  const deltas: FormattedDelta[] = [
    { label: "Resultado", value: isVictory ? "VitÃ³ria Plena" : "Empate Tenso", isPositive: isVictory },
    { label: "ForÃ§a Efetiva", value: `${effectivePlayer} vs ${effectiveRival}`, isPositive: isVictory },
    { label: "Moral", value: isVictory ? "+6 Moral" : "-5 Moral", isPositive: isVictory },
    { label: "Despesas Gastas", value: `R$ ${cost.toLocaleString()}`, isPositive: cost === 0 },
    { label: "DeserÃ§Ã£o de SÃ³cios", value: `-${desertion} sÃ³cios`, isPositive: desertion < 30 },
  ];

  return {
    effectiveForcePlayer: effectivePlayer,
    effectiveForceRival: effectiveRival,
    scorePlayer,
    scoreRival,
    isVictory,
    isAllyGame: false,
    statusTitle: isVictory ? "FESTA E DOMÃNIO ABSOLUTO EM CASA" : "PRESSÃƒO E ATRITO DE ENTORNO",
    memberDesertion: desertion,
    medicalExpenses: medical,
    bannerLost: false,
    extraCost: cost,
    mpPenalty,
    moralChange: isVictory ? 6 : -5,
    chronicleText: `A bateria ditou o ritmo ininterrupto do primeiro ao Ãºltimo minuto. O setor local vibrou com a recepÃ§Ã£o de fumaÃ§a e as faixas estendidas, sustentando o espetÃ¡culo visual e a seguranÃ§a dos portÃµes de acesso.`,
    formattedDeltas: deltas,
  };
}

export function calculateFormulaAwayGame(
  stats: TorcidaStats,
  state: StateTrackers,
  setup: AwayGameSetup
): KeyGameResult {
  let logisticsBonus = 40;
  let cost = 0;
  let mpPenalty = 5;

  if (setup.distanceType === "SHORT") {
    if (setup.shortTransport === "BONDE_TREM_METRO") {
      logisticsBonus += 20;
      mpPenalty += 15;
    } else if (setup.shortTransport === "ARRANCADA_CAMINHADA") {
      logisticsBonus += 25;
      mpPenalty += 10;
    }
  } else {
    if (setup.longTransport === "FRETADO_BARATO") {
      cost += 6000;
      logisticsBonus += 15;
    } else if (setup.longTransport === "ONIBUS_EXECUTIVO") {
      cost += 14000;
      logisticsBonus += 35;
    }
  }

  if (setup.armaments === "SINALIZADORES_BATEDORES") {
    cost += 4000;
    logisticsBonus += 20;
  }

  if (setup.tactic === "ROTA_SEM_ESCOLTA") {
    logisticsBonus += 15;
    mpPenalty += 20;
  }

  const rngPlayer = (Math.random() - 0.5) * 30;
  const effectivePlayer = Math.round(
    stats.poder_pista * 0.35 +
      stats.caravana * 0.15 +
      state.moral * 0.2 +
      logisticsBonus * 0.15 +
      rngPlayer
  );

  const effectiveRival = Math.round(70 + (Math.random() - 0.5) * 30);
  const isVictory = effectivePlayer >= effectiveRival;

  const scorePlayer = isVictory ? Math.floor(Math.random() * 2 + 1) : Math.floor(Math.random() * 2);
  const scoreRival = isVictory ? Math.floor(Math.random() * scorePlayer) : scorePlayer + 1;

  const desertion = isVictory ? Math.floor(Math.random() * 30 + 10) : Math.floor(Math.random() * 80 + 40);
  const medical = isVictory ? 500 : Math.floor(Math.random() * 3000 + 1200);
  const bannerLost = !isVictory && Math.random() < 0.15;

  const deltas: FormattedDelta[] = [
    { label: "Status Pista", value: isVictory ? "VitÃ³ria na Rua" : "Recuo EstratÃ©gico", isPositive: isVictory },
    { label: "ForÃ§a Pista", value: `${effectivePlayer} vs ${effectiveRival}`, isPositive: isVictory },

    { label: "DeserÃ§Ã£o de SÃ³cios", value: `-${desertion} sÃ³cios`, isPositive: desertion < 30 },
  ];

  return {
    effectiveForcePlayer: effectivePlayer,
    effectiveForceRival: effectiveRival,
    scorePlayer,
    scoreRival,

    isAllyGame: false,

    mpPenalty,
